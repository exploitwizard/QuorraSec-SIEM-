STEP 2

Integrate the SDK

Add Courra-Sec to your website or app using one of the methods below.

<!-- Add before </body> in your HTML -->
<script src="{{ siem_url }}/courra-sec-sdk.js"></script>
<script>
  Courra-SecSDK.init({
    url:    '{{ siem_url }}',
    apiKey: '{{ org.api_key }}',
    debug:  false,
  });
</script>
# Install the package
npm install courra-sec-sdk

# In your JavaScript / TypeScript
import Courra-SecSDK from 'courra-sec-sdk';

Courra-SecSDK.init({
  url:    '{{ siem_url }}',
  apiKey: '{{ org.api_key }}',
});
# Add to your .env file
COURRA-SEC_INGEST_URL={{ siem_url }}
COURRA-SEC_API_KEY={{ org.api_key }}

# Python example
import os, requests
requests.post(
    os.environ['COURRA-SEC_INGEST_URL'] + '/ingest',
    json=event,
    headers={'X-API-Key': os.environ['COURRA-SEC_API_KEY']}
)
